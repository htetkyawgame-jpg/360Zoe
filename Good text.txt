    <script>
        let accounts = JSON.parse(localStorage.getItem('pubg_accounts_v2')) || [];
        let currentFilter = 'all';
        let currentImagesBase64 = [];
        let storeCardIndexes = {};

        let activeModalImages = [];
        let activeModalIndex = 0;

        document.getElementById('form-image-file').addEventListener('change', function(e) {
            const files = Array.from(e.target.files);
            if (files.length > 0) {
                let loadedCount = 0;
                files.forEach(file => {
                    const reader = new FileReader();
                    reader.onload = function(event) {
                        currentImagesBase64.push(event.target.result);
                        loadedCount++;
                        if (loadedCount === files.length) {
                            renderImagePreviews();
                        }
                    };
                    reader.readAsDataURL(file);
                });
            }
        });

        function renderImagePreviews() {
            const container = document.getElementById('image-preview-container');
            container.innerHTML = '';
            currentImagesBase64.forEach((imgSrc, idx) => {
                const div = document.createElement('div');
                div.className = 'relative w-14 h-14 bg-pubg-dark rounded border border-pubg-border overflow-hidden';
                div.innerHTML = `
                    <img src="${imgSrc}" class="w-full h-full object-cover">
                    <button type="button" onclick="removeImage(${idx})" class="absolute top-0 right-0 bg-red-600 text-white text-xs w-4 h-4 flex items-center justify-center"><i class="fa-solid fa-xmark"></i></button>
                `;
                container.appendChild(div);
            });
        }

        function removeImage(index) {
            currentImagesBase64.splice(index, 1);
            renderImagePreviews();
        }

        function saveToLocalStorage() {
            localStorage.setItem('pubg_accounts_v2', JSON.stringify(accounts));
        }

        function switchTab(tab) {
            document.getElementById('view-store').classList.toggle('hidden', tab !== 'store');
            document.getElementById('view-admin').classList.toggle('hidden', tab !== 'admin');
            
            document.getElementById('btn-view-store').className = tab === 'store' 
                ? "px-4 py-2 rounded-lg font-bold text-sm flex items-center space-x-2 transition bg-pubg-gold text-black shadow-md shadow-pubg-gold/20"
                : "px-4 py-2 rounded-lg font-bold text-sm flex items-center space-x-2 transition bg-pubg-border text-gray-300 hover:text-white hover:bg-gray-700";

            document.getElementById('btn-view-admin').className = tab === 'admin'
                ? "px-4 py-2 rounded-lg font-bold text-sm flex items-center space-x-2 transition bg-pubg-gold text-black shadow-md shadow-pubg-gold/20"
                : "px-4 py-2 rounded-lg font-bold text-sm flex items-center space-x-2 transition bg-pubg-border text-gray-300 hover:text-white hover:bg-gray-700";
        }

        function setFilter(filter) {
            currentFilter = filter;
            ['all', 'available', 'sold'].forEach(f => {
                document.getElementById(`filter-${f}`).className = f === filter 
                    ? "flex-1 md:flex-none px-5 py-1.5 rounded-md text-sm font-bold transition bg-pubg-gold text-black shadow"
                    : "flex-1 md:flex-none px-5 py-1.5 rounded-md text-sm font-bold text-gray-400 hover:text-white transition";
            });
            renderStore();
        }

        function changeCardImg(accId, direction, totalImgs, e) {
            e.stopPropagation();
            if (!storeCardIndexes[accId]) storeCardIndexes[accId] = 0;
            storeCardIndexes[accId] = (storeCardIndexes[accId] + direction + totalImgs) % totalImgs;
            
            const acc = accounts.find(a => a.id === accId);
            if (acc && acc.images && acc.images[storeCardIndexes[accId]]) {
                document.getElementById(`img-card-${accId}`).src = acc.images[storeCardIndexes[accId]];
                document.getElementById(`counter-card-${accId}`).innerText = `${storeCardIndexes[accId] + 1}/${totalImgs}`;
            }
        }

        function openModal(accId) {
            const acc = accounts.find(a => a.id === accId);
            const defaultImg = 'https://images.unsplash.com/photo-1542751371-adc38448a05e?auto=format&fit=crop&w=600&q=80';
            activeModalImages = (acc && acc.images && acc.images.length > 0) ? acc.images : [defaultImg];
            activeModalIndex = storeCardIndexes[accId] || 0;

            updateModal();
            document.getElementById('image-modal').classList.remove('hidden');
        }

        function closeModal() {
            document.getElementById('image-modal').classList.add('hidden');
        }

        function updateModal() {
            document.getElementById('modal-img').src = activeModalImages[activeModalIndex];
            document.getElementById('modal-counter').innerText = `${activeModalIndex + 1} / ${activeModalImages.length}`;
        }

        function prevModalImg() {
            activeModalIndex = (activeModalIndex - 1 + activeModalImages.length) % activeModalImages.length;
            updateModal();
        }

        function nextModalImg() {
            activeModalIndex = (activeModalIndex + 1) % activeModalImages.length;
            updateModal();
        }

        function renderStore() {
            const grid = document.getElementById('account-grid');
            const search = document.getElementById('store-search').value.toLowerCase().trim();
            grid.innerHTML = '';

            const filtered = accounts.filter(acc => {
                const matchesFilter = currentFilter === 'all' || acc.status === currentFilter;
                const matchesSearch = acc.code.toLowerCase().includes(search) || acc.title.toLowerCase().includes(search);
                return matchesFilter && matchesSearch;
            });

            document.getElementById('store-empty').classList.toggle('hidden', filtered.length > 0);

            filtered.forEach(acc => {
                const isSold = acc.status === 'sold';
                const defaultImg = 'https://images.unsplash.com/photo-1542751371-adc38448a05e?auto=format&fit=crop&w=600&q=80';
                
                let imgList = acc.images;
                if (!imgList || imgList.length === 0) {
                    imgList = acc.image ? [acc.image] : [defaultImg];
                }

                if (storeCardIndexes[acc.id] === undefined) {
                    storeCardIndexes[acc.id] = 0;
                }
                const currentIdx = storeCardIndexes[acc.id];

                const card = document.createElement('div');
                card.className = `bg-pubg-card border border-pubg-border rounded-2xl overflow-hidden shadow-lg flex flex-col justify-between ${isSold ? 'opacity-75' : ''}`;
                card.innerHTML = `
                    <div>
                        <div class="relative h-56 w-full bg-pubg-dark overflow-hidden cursor-pointer" onclick="openModal('${acc.id}')">
                            <img id="img-card-${acc.id}" src="${imgList[currentIdx]}" class="w-full h-full object-cover transition-all duration-300">
                            <div class="absolute top-3 left-3 z-10">
                                ${isSold 
                                    ? '<span class="bg-red-600 text-white font-black text-xs px-3 py-1 rounded-md uppercase">SOLD OUT</span>' 
                                    : '<span class="bg-green-500 text-black font-extrabold text-xs px-3 py-1 rounded-md uppercase">AVAILABLE</span>'}
                            </div>

                            ${imgList.length > 1 ? `
                                <button onclick="changeCardImg('${acc.id}', -1,${imgList.length}, event)" class="absolute left-2 top-1/2 -translate-y-1/2 bg-black/60 text-white px-2 py-1 rounded-full hover:bg-pubg-gold hover:text-black z-10"><i class="fa-solid fa-chevron-left text-xs"></i></button>
                                <button onclick="changeCardImg('${acc.id}', 1,${imgList.length}, event)" class="absolute right-2 top-1/2 -translate-y-1/2 bg-black/60 text-white px-2 py-1 rounded-full hover:bg-pubg-gold hover:text-black z-10"><i class="fa-solid fa-chevron-right text-xs"></i></button>
                                <div id="counter-card-${acc.id}" class="absolute bottom-2 right-2 bg-black/70 text-white text-[10px] px-2 py-0.5 rounded font-bold">
                                    ${currentIdx + 1}/${imgList.length}
                                </div>
                            ` : ''}
                        </div>
                        <div class="p-5 space-y-2">
                            <div class="flex justify-between items-center">
                                <h3 class="teko-font text-2xl font-bold text-white">${acc.code}</h3>
                                <span class="text-pubg-gold font-bold text-lg">${acc.price}</span>
                            </div>
                            <p class="text-xs text-gray-300 bg-pubg-dark/50 p-2.5 rounded-lg border border-pubg-border/50 leading-relaxed">${acc.title}</p>
                        </div>
                    </div>
                `;
                grid.appendChild(card);
            });
        }

        function renderAdminTable() {
            const tbody = document.getElementById('admin-table-body');
            tbody.innerHTML = '';

            if (accounts.length === 0) {
                tbody.innerHTML = `<tr><td colspan="6" class="p-4 text-center text-gray-500">စာရင်း မရှိသေးပါ။</td></tr>`;
                return;
            }

            accounts.forEach(acc => {
                const imgCount = acc.images ? acc.images.length : (acc.image ? 1 : 0);
                const tr = document.createElement('tr');
                tr.innerHTML = `
                    <td class="p-3 font-bold text-white">${acc.code}</td>
                    <td class="p-3 text-xs text-gray-300 truncate max-w-xs">${acc.title}</td>
                    <td class="p-3 font-semibold text-pubg-gold">${acc.price}</td>
                    <td class="p-3 font-semibold text-gray-300"><i class="fa-solid fa-images text-pubg-gold mr-1"></i> ${imgCount} ပုံ</td>
                    <td class="p-3">
                        <button onclick="toggleStatus('${acc.id}')" class="text-xs font-bold px-3 py-1 rounded-full ${acc.status === 'available' ? 'bg-green-500/10 text-green-400 border border-green-500/30' : 'bg-red-500/10 text-red-400 border border-red-500/30'}">
                            ${acc.status === 'available' ? 'Available' : 'Sold Out'}
                        </button>
                    </td>
                    <td class="p-3 text-right space-x-2">
                        <button onclick="editAccount('${acc.id}')" class="text-blue-400"><i class="fa-solid fa-pen-to-square"></i></button>
                        <button onclick="deleteAccount('${acc.id}')" class="text-red-400"><i class="fa-solid fa-trash"></i></button>
                    </td>
                `;
                tbody.appendChild(tr);
            });
        }

        function handleFormSubmit(e) {
            e.preventDefault();

            const id = document.getElementById('form-id').value;
            const code = document.getElementById('form-code').value.trim();
            const price = document.getElementById('form-price').value.trim();
            const title = document.getElementById('form-title-text').value.trim();
            const status = document.getElementById('form-status').value;

            if (id) {
                const index = accounts.findIndex(a => a.id === id);
                if (index !== -1) {
                    accounts[index] = { 
                        ...accounts[index], 
                        code, price, title, status,
                        images: currentImagesBase64.length > 0 ? currentImagesBase64 : (accounts[index].images || [])
                    };
                }
            } else {
                accounts.unshift({
                    id: 'acc_' + Date.now(),
                    code, price, title, status,
                    images: [...currentImagesBase64]
                });
            }

            saveToLocalStorage();
            resetForm();
            renderStore();
            renderAdminTable();
        }

        function editAccount(id) {
            const acc = accounts.find(a => a.id === id);
            if (!acc) return;

            document.getElementById('form-id').value = acc.id;
            document.getElementById('form-code').value = acc.code;
            document.getElementById('form-price').value = acc.price;
            document.getElementById('form-title-text').value = acc.title;
            document.getElementById('form-status').value = acc.status;
            
            currentImagesBase64 = acc.images ? [...acc.images] : (acc.image ? [acc.image] : []);
            renderImagePreviews();

            document.getElementById('form-title').innerText = 'အကောင့် ပြင်ဆင်ရန်';
            document.getElementById('form-submit-btn').innerText = 'ပြင်ဆင်ချက် သိမ်းမည်';
        }

        function toggleStatus(id) {
            const acc = accounts.find(a => a.id === id);
            if (acc) {
                acc.status = acc.status === 'available' ? 'sold' : 'available';
                saveToLocalStorage();
                renderStore();
                renderAdminTable();
            }
        }

        function deleteAccount(id) {
            if (confirm('ဤအကောင့်အား ဖျက်ရန် သေချာပါသလား။')) {
                accounts = accounts.filter(a => a.id !== id);
                saveToLocalStorage();
                renderStore();
                renderAdminTable();
            }
        }

        }
